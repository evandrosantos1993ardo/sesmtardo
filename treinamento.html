📋 Código do Apps Script

Cole no Apps Script (Extensões → Apps Script) substituindo SEU_ID pelo ID da sua planilha:

const SHEET_ID = 'SEU_ID_AQUI'; function doGet(e) { const a = e.parameter.action; const ss = SpreadsheetApp.openById(SHEET_ID); function tab(name, headers) { let s = ss.getSheetByName(name); if (!s) { s = ss.insertSheet(name); s.appendRow(headers); } return s; } const SC = tab('COLABORADORES',['ID','Nome','Matricula','Cargo','Setor','Funcao','Tipo','Telefone','Admissao','Status']); const ST = tab('TREINAMENTOS',['ID','NomeColab','Matricula','Treinamento','Tipo','DataReal','Validade','DataVenc','Status','Instrutor','Obs']); const SN = tab('CNH',['ID','NomeMotorista','Matricula','Categoria','NumeroCNH','ValidadeCNH','DataToxico','VencToxico','StatusCNH','StatusToxico']); const SL = tab('LOG',['DataHora','Tipo','Descricao']); if (a==='getColaboradores') return out({ok:true,data:SC.getDataRange().getValues()}); if (a==='getTreinamentos') return out({ok:true,data:ST.getDataRange().getValues()}); if (a==='getCNH') return out({ok:true,data:SN.getDataRange().getValues()}); if (a==='getLog') return out({ok:true,data:SL.getDataRange().getValues()}); if (a==='addColaborador') { SC.appendRow([e.parameter.id,e.parameter.nome,e.parameter.matricula,e.parameter.cargo,e.parameter.setor,e.parameter.funcao,e.parameter.tipo,e.parameter.telefone,e.parameter.admissao,e.parameter.status||'Ativo']); return out({ok:true}); } if (a==='addTreinamento') { ST.appendRow([e.parameter.id,e.parameter.nomeColab,e.parameter.matricula,e.parameter.treinamento,e.parameter.tipo,e.parameter.dataReal,e.parameter.validade,e.parameter.dataVenc,e.parameter.status,e.parameter.instrutor,e.parameter.obs||'']); return out({ok:true}); } if (a==='addCNH') { SN.appendRow([e.parameter.id,e.parameter.nomeMotorista,e.parameter.matricula,e.parameter.categoria,e.parameter.numeroCNH,e.parameter.validadeCNH,e.parameter.dataToxico,e.parameter.vencToxico,e.parameter.statusCNH,e.parameter.statusToxico]); return out({ok:true}); } if (a==='addLog') { SL.appendRow([e.parameter.dataHora,e.parameter.tipo,e.parameter.descricao]); return out({ok:true}); } if (a==='deleteColab') { delRow(SC,e.parameter.id); return out({ok:true}); } if (a==='deleteTrein') { delRow(ST,e.parameter.id); return out({ok:true}); } if (a==='deleteCNH') { delRow(SN,e.parameter.id); return out({ok:true}); } return out({ok:false}); } function delRow(sheet,id){ const rows=sheet.getDataRange().getValues(); for(let i=1;i<rows.length;i++){if(String(rows[i][0])===String(id)){sheet.deleteRow(i+1);return;}} } function out(obj){ return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); }
