📋 Configuração do Google Apps Script

Siga os passos abaixo para configurar o backend no Google Drive:

1. Criar a Planilha

Acesse sheets.google.com, crie uma planilha nova chamada "Controle EPI" e copie o ID da URL.


2. Criar o Apps Script

Na planilha, clique em Extensões → Apps Script. Cole o código abaixo:

const SHEET_ID = 'SEU_SHEET_ID_AQUI';
const SHEET_NAME = 'EPI';

function doGet(e) {
  const action = e.parameter.action;
  const ss = SpreadsheetApp.openById(SHEET_ID);
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow(['Código','Data Cadastro','Nome','Descrição',
      'CA','Estoque Mínimo','Saldo Atual']);
  }
  if (action === 'getAll') {
    const data = sheet.getDataRange().getValues();
    return ContentService.createTextOutput(
      JSON.stringify({ok:true,data:data})
    ).setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput('{}');
}

function doPost(e) {
  const body = JSON.parse(e.postData.contents);
  const ss = SpreadsheetApp.openById(SHEET_ID);
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) sheet = ss.insertSheet(SHEET_NAME);
  const { action, data, codigo, quantidade } = body;

  if (action === 'add') {
    sheet.appendRow([data.codigo, data.dataCadastro, data.nome,
      data.descricao, data.ca, data.estoqueMin, data.saldoAtual]);
    return ok();
  }
  if (action === 'saida' || action === 'entrada') {
    const rows = sheet.getDataRange().getValues();
    for (let i = 1; i < rows.length; i++) {
      if (rows[i][0] == codigo) {
        const col = 7;
        const atual = rows[i][6];
        const novo = action==='saida'
          ? Math.max(0, atual - quantidade)
          : atual + quantidade;
        sheet.getRange(i+1, col).setValue(novo);
        return ok();
      }
    }
  }
  if (action === 'delete') {
    const rows = sheet.getDataRange().getValues();
    for (let i = 1; i < rows.length; i++) {
      if (rows[i][0] == codigo) {
        sheet.deleteRow(i+1); return ok();
      }
    }
  }
  return ok();
}
function ok() {
  return ContentService.createTextOutput(
    JSON.stringify({ok:true})
  ).setMimeType(ContentService.MimeType.JSON);
}
3. Publicar como Web App

Clique em Implantar → Nova implantação. Tipo: App da Web. Executar como: Eu. Quem tem acesso: Qualquer pessoa. Copie a URL gerada.
